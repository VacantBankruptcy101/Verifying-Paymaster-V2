paymasterAndData Signing Spec
// Hash signed by the backend signer (EIP-191 personal_sign or raw secp256k1)
const hash = ethers.utils.keccak256(ethers.utils.solidityPack(
  ['bytes32','uint256','address','uint8','uint48','uint48','address'],
  [userOpHash, chainId, paymasterAddr, mode, validUntil, validAfter, tokenAddr]
));
const sig = await signer.signMessage(ethers.utils.arrayify(hash));

// Pack into paymasterAndData (20 + 1 + 6 + 6 + 20 + 65 = 118 bytes)
const paymasterAndData = ethers.utils.hexConcat([
  paymasterAddr,         // [0:20]
  [mode],                // [20]
  packUint48(validUntil),// [21:27]
  packUint48(validAfter),// [27:33]
  tokenAddr,             // [33:53]
  sig                    // [53:118] ECDSA 65 bytes
]);
