JS SDK — Sign a UserOperation
const { buildPaymasterData, MODE_SPONSORED } = require('./scripts/paymasterSigner');

const paymasterAndData = await buildPaymasterData({
  paymasterAddress: '0x4a9e…c7f2',
  userOpHash,           // from EntryPoint
  signerWallet,         // ethers.Signer (hot-key backend)
  mode: MODE_SPONSORED,
  validUntilSec: Math.floor(Date.now() / 1000) + 300,
});

// Attach to UserOperation before submitting to bundler
userOp.paymasterAndData = paymasterAndData;
